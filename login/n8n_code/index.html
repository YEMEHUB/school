  const n = $json.숫자;
  if (n < 2) return false;
  for (let i = 2; i <= Math.sqrt(n); i++) {
    if (n % i === 0) return false;
  }
  return true;
})() }}

🤖 AI 요약 실습 원문

인공지능 기술은 과거에는 인간이 직접 수행해야 한다고 여겨졌던 다양한 인지적 활동을 컴퓨터가 처리할 수 있도록 발전해 왔다. 초기의 인공지능은 개발자가 미리 정해 놓은 규칙에 따라 제한된 문제를 해결하는 방식이 중심이었지만, 컴퓨터의 연산 능력이 향상되고 대규모 데이터를 수집할 수 있는 환경이 만들어지면서 데이터로부터 일정한 패턴을 찾아내는 머신러닝 기술이 빠르게 발전하였다. 특히 딥러닝은 여러 층으로 구성된 인공신경망을 이용하여 복잡한 데이터의 특징을 단계적으로 학습하는 방법으로, 이미지 인식이나 음성 인식, 자연어 처리와 같은 분야에서 높은 성능을 보여 왔다. 최근에는 방대한 양의 텍스트를 학습한 생성형 인공지능이 등장하면서 단순히 정보를 분류하거나 예측하는 수준을 넘어 새로운 글과 이미지, 음악, 프로그램 코드 등을 생성하는 것도 가능해졌다. 그러나 인공지능이 만들어 내는 결과가 항상 정확하거나 객관적인 것은 아니다. 인공지능은 학습 과정에서 사용된 데이터의 특징과 한계를 함께 반영할 수 있기 때문에 데이터에 편향된 정보가 포함되어 있다면 결과에도 그러한 편향이 나타날 가능성이 있다. 또한 생성형 인공지능은 질문에 대한 정확한 정보를 가지고 있지 않은 상황에서도 문맥상 자연스러워 보이는 내용을 만들어 낼 수 있는데, 이처럼 사실과 다른 정보를 마치 사실인 것처럼 생성하는 현상을 흔히 ‘환각(Hallucination)’이라고 부른다. 따라서 인공지능의 답변을 그대로 받아들이기보다는 출처를 확인하거나 다른 자료와 비교하면서 정보의 신뢰성을 판단하는 과정이 필요하다. 인공지능의 확산은 인간의 직업과 역할에도 변화를 가져오고 있다. 반복적으로 처리해야 하는 업무나 많은 자료를 빠르게 분석해야 하는 작업은 인공지능을 활용하여 효율성을 높일 수 있지만, 이것이 반드시 특정 직업 전체가 사라진다는 의미는 아니다. 하나의 직업은 여러 종류의 업무로 구성되어 있으며, 그중 일부는 자동화하기 쉽지만 사람과의 의사소통, 새로운 문제의 정의, 상황에 따른 판단처럼 인간의 경험과 맥락 이해가 중요한 업무는 여전히 사람의 역할이 크게 요구될 수 있다.

🔒 개인정보 가리기

const name = $json['이름'] || '';
const phone = $json['전화번호'] || '';

return {
  json: {
    가린이름: name.charAt(0) + '○'.repeat(Math.max(name.length - 1, 1)),
    가린전화: phone.replace(/\d{4}$/, '****'),
    안내: '개인정보는 일부를 가려서 보관해요'
  }
};
